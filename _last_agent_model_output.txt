from __future__ import annotations

from typing import List

from .types import Candidate


class StrategyV1:
    """
    Deterministic, long-only strategy (v1).

    Rules (configurable via constructor or Settings if extended):
    - BUY candidate if:
      - last_close > SMA(20)
      - RSI(14) between [rsi_lower, rsi_upper] (defaults 40..70)

    generate(...) expects `data.get_latest(symbol)` to return a mapping with keys:
      - "close"
      - "sma20"
      - "rsi14"

    The score is a simple heuristic combining distance above SMA and closeness
    of RSI to the midpoint of the allowed RSI range.
    """

    def __init__(self, rsi_lower: float = 40.0, rsi_upper: float = 70.0):
        self.rsi_lower = float(rsi_lower)
        self.rsi_upper = float(rsi_upper)

    def generate(self, symbols: List[str], data: "DataClient", cfg: "Settings") -> List[Candidate]:
        candidates: List[Candidate] = []
        for sym in symbols:
            info = data.get_latest(sym)
            if not info:
                continue
            try:
                close = float(info["close"])
                sma20 = float(info["sma20"])
                rsi = float(info["rsi14"])
            except Exception:
                # missing or invalid data for this symbol
                continue

            # Rule checks
            if not (close > sma20):
                continue
            if not (self.rsi_lower <= rsi <= self.rsi_upper):
                continue

            # Score calculation
            distance = (close - sma20) / sma20 if sma20 != 0 else 0.0

            rsi_mid = (self.rsi_lower + self.rsi_upper) / 2.0
            # Normalize RSI closeness to [0,1], where 1 is at midpoint and 0 at edges or beyond.
            half_range = (self.rsi_upper - self.rsi_lower) / 2.0 or 1.0
            rsi_score = 1.0 - (abs(rsi - rsi_mid) / half_range)
            rsi_score = max(0.0, min(1.0, rsi_score))

            score = max(0.0, distance * 0.7 + rsi_score * 0.3)

            reason = f"Close {close:.2f} above SMA20 {sma20:.2f}, RSI {rsi:.1f}"
            snapshot = {"close": close, "sma20": sma20, "rsi14": rsi}

            candidates.append(Candidate(symbol=sym, score=score, reason=reason, snapshot=snapshot))

        candidates.sort(key=lambda c: c.score, reverse=True)
        return candidates

import pytest

from tradingbot.strategy.strategy_v1 import StrategyV1


class FakeDataClient:
    def __init__(self, mapping):
        # mapping: symbol -> dict with keys "close", "sma20", "rsi14"
        self.mapping = mapping

    def get_latest(self, symbol):
        return self.mapping.get(symbol, {})


class FakeSettings:
    pass


def test_no_candidates_when_rules_fail():
    data = FakeDataClient(
        {
            "AAA": {"close": 90, "sma20": 100, "rsi14": 50},  # close <= sma
            "BBB": {"close": 110, "sma20": 100, "rsi14": 30},  # rsi too low
            "CCC": {"close": 110, "sma20": 100, "rsi14": 80},  # rsi too high
        }
    )
    strat = StrategyV1()
    candidates = strat.generate(["AAA", "BBB", "CCC"], data, FakeSettings())
    assert candidates == []


def test_candidate_fields_filled_when_rules_pass():
    data = FakeDataClient(
        {
            "GOOD": {"close": 120.0, "sma20": 100.0, "rsi14": 55.0},
        }
    )
    strat = StrategyV1()
    candidates = strat.generate(["GOOD"], data, FakeSettings())
    assert len(candidates) == 1
    c = candidates[0]
    assert c.symbol == "GOOD"
    assert isinstance(c.score, float)
    assert c.score > 0
    assert "Close" in c.reason
    assert isinstance(c.snapshot, dict)
    assert c.snapshot["close"] == 120.0
    assert c.snapshot["sma20"] == 100.0
    assert c.snapshot["rsi14"] == 55.0
