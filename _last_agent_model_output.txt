from typing import List, Optional


def sma(values: List[float], window: int) -> List[Optional[float]]:
    """Simple moving average. Returns list of same length with None for
    positions where there isn't enough history."""
    if window <= 0:
        raise ValueError("window must be > 0")
    n = len(values)
    if n == 0:
        return []
    result: List[Optional[float]] = [None] * n
    if window > n:
        return result
    # compute first window sum
    window_sum = sum(values[:window])
    result[window - 1] = window_sum / window
    for i in range(window, n):
        window_sum += values[i] - values[i - window]
        result[i] = window_sum / window
    return result


def rsi(values: List[float], window: int = 14) -> List[Optional[float]]:
    """Relative Strength Index using Wilder's smoothing.
    Returns a list of same length with None for indices that don't have enough history.
    RSI values are in range [0, 100]. For flat periods (no gains and no losses) we return 50.0.
    """
    if window <= 0:
        raise ValueError("window must be > 0")
    n = len(values)
    if n == 0:
        return []
    # Need at least window+1 values to compute first RSI
    if n <= window:
        return [None] * n
    # deltas length n-1, delta[i] = values[i+1] - values[i]
    deltas = [values[i] - values[i - 1] for i in range(1, n)]
    # initial average gains/losses from first `window` deltas
    gains = [d if d > 0 else 0.0 for d in deltas[:window]]
    losses = [-d if d < 0 else 0.0 for d in deltas[:window]]
    avg_gain = sum(gains) / window
    avg_loss = sum(losses) / window

    def compute_rsi_from_avgs(ag: float, al: float) -> float:
        if ag == 0.0 and al == 0.0:
            return 50.0
        if al == 0.0:
            return 100.0
        rs = ag / al
        return 100.0 - (100.0 / (1.0 + rs))

    result: List[Optional[float]] = [None] * n
    # first RSI value sits at index `window`
    result[window] = compute_rsi_from_avgs(avg_gain, avg_loss)
    # subsequent values use Wilder smoothing
    for idx in range(window + 1, n):
        delta = deltas[idx - 1]
        gain = delta if delta > 0 else 0.0
        loss = -delta if delta < 0 else 0.0
        avg_gain = (avg_gain * (window - 1) + gain) / window
        avg_loss = (avg_loss * (window - 1) + loss) / window
        result[idx] = compute_rsi_from_avgs(avg_gain, avg_loss)
    return result


def trend_up(values: List[float], lookback: int = 5) -> bool:
    """Simple trend check: return True if last value > value[-lookback].
    If there isn't enough history, return False."""
    if lookback <= 0:
        raise ValueError("lookback must be > 0")
    if len(values) < lookback:
        return False
    return values[-1] > values[-lookback]
# FILE: tests/test_indicators.py
import pytest

from tradingbot.indicators import rsi, sma, trend_up


def test_sma_basic():
    values = [1, 2, 3, 4, 5]
    assert sma(values, 3) == [None, None, 2.0, 3.0, 4.0]


def test_sma_window_one_and_short():
    values = [5, 6, 7]
    assert sma(values, 1) == [5.0, 6.0, 7.0]
    assert sma(values, 5) == [None, None, None]


def test_rsi_all_increase():
    # 1..15 (length 15) with window 14 -> there is exactly one RSI value at index 14
    values = list(range(1, 16))
    out = rsi(values, 14)
    assert out[:14] == [None] * 14
    assert out[14] == pytest.approx(100.0)


def test_rsi_flat_series():
    values = [1, 1, 1, 1, 1]
    out = rsi(values, 3)
    # first three positions have insufficient history
    assert out[:3] == [None, None, None]
    # with no gains and no losses RSI should be 50.0
    assert out[3] == pytest.approx(50.0)
    assert out[4] == pytest.approx(50.0)


def test_rsi_short_series():
    # not enough data to compute RSI
    assert rsi([1, 2, 3], 5) == [None, None, None]


def test_trend_up_behaviour():
    assert trend_up([1, 2, 3, 4, 5], 5) is True
    assert trend_up([5, 4, 3, 2, 1], 5) is False
    # insufficient history returns False
    assert trend_up([1, 2, 3], 5) is False
    # explicit lookback
    assert trend_up([1, 2, 3, 4, 3], 4) is False
    assert trend_up([1, 2, 3, 4, 5], 3) is True
